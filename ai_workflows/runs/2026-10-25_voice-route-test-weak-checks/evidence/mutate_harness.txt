export, repo, test = sys.argv[1], sys.argv[2], sys.argv[3]
log = os.environ['OPENER_LOG']
for name in sys.argv[4:]:
    target = None
    if name != 'none':
        rel, old, new = MUT[name]
        target = pathlib.Path(export, rel)
        src = target.read_text()
        assert src.count(old) == 1, f'{name}: anchor found {src.count(old)} times'
        target.write_text(src.replace(old, new))
    open(log, 'w').close()
    try:
        out = subprocess.run(['pnpm', 'test', '--', f'--only={test}'], cwd=export, capture_output=True, text=True, timeout=600).stdout
    finally:
        if target is not None:
            target.write_bytes(subprocess.run(['git', '-C', repo, 'show', f'HEAD:{rel}'], capture_output=True, check=True).stdout)
    verdict = next((l.split()[0] for l in out.splitlines() if l.strip().endswith('ms') and test in l), '?')
    fails = list(dict.fromkeys(l.strip() for l in out.splitlines() if 'FAIL:' in l))
    calls = sum(1 for _ in open(log))
    print(f'{name:26} {verdict:8} opener_calls={calls} failing_checks={len(fails)}')
    for f in fails: print('      ', f[:150])
